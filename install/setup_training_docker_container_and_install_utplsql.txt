# Download and create docker container
podman run -d --name deathstar -p 1521:1521 -e ORACLE_PWD=oracle container-registry.oracle.com/database/express:21.3.0-xe

# Install git in the container
podman exec -it -u root deathstar bash

yml install git

exit

cd source
